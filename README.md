# Comandos Git

# Inicia um novo repositório Git localmente
git init

# Clona um repositório remoto para a sua máquina
git clone [url-do-repositorio]

# Verifica o status dos arquivos (modificados, adicionados, não rastreados)
git status

# Adiciona um arquivo específico para a área de preparação (staging)
git add nome-do-arquivo.ext

# Adiciona todos os arquivos modificados de uma vez
git add .

# Salva as alterações com uma mensagem descritiva
git commit -m "Sua mensagem descritiva aqui"

# Exibe o histórico de commits
git log --oneline
